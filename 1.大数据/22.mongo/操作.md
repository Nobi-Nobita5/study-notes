db.deviceFingerprint.aggregate([
  {
        $match: {
            "method_id": "multiPlatformDetection"  // 可选：只看faceSearch的数据
        }
  },
  {
    		$project: {
      			fields: { $objectToArray: "$$ROOT" }
    		}
  },
  {
    $unwind: "$fields"
  },
  {
    $group: {
      _id: "$fields.k",  // 按字段名分组
      types: { $addToSet: { $type: "$fields.v" } },  // 收集所有出现过的类型
    }
  },
  {
    $sort: { _id: 1 }  // 按字段名排序
  }
])

```
//显示mongo文档中某个字段（json报文）出现过的类型
db.advanceLog.aggregate([
    {
        $match: {
            "method_id": "multiPlatformDetection"  // 可选：只看faceSearch的数据
        }
    },
    {
        $group: {
            _id: { $type: "$return_data.data" },
            count: { $sum: 1 },
            examples: { $push: "$_id" }  // 收集一些示例文档ID
        }
    },
    {
        $project: {
            dataType: "$_id",
            count: 1,
            examples: { $slice: ["$examples", 3] },  // 只显示前3个示例
            _id: 0
        }
    },
    {
        $sort: { count: -1 }
    }
])

//再看object报文是否四种类型都有：array,[],{},null
db.advanceLog.find({
    "method_id": "multiPlatformDetection",
    "return_data.data": { 
        $exists: true, 
        $type: "object",
        $not: { $type: "array" }  // 明确排除数组
    }
})

db.advanceLog.find({
    "method_id": "multiPlatformDetection",
    "return_data.data": { 
        $exists: true, 
        $type: "array"  //数组
    }
})

db.advanceLog.findOne({
    "method_id": "multiPlatformDetection",
    "return_data.data": [] //空数组
})

db.advanceLog.findOne({
    "method_id": "multiPlatformDetection",
    "return_data.data": null //空值不用单独spark sql处理
})
```
